PREFIX knora-base: <http://www.knora.org/ontology/knora-base#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?resource ?labelOrIri
WHERE {
  {
    {
      ?resource a <http://www.knora.org/ontology/0001/anything#Thing> ;
        knora-base:attachedToProject <http://rdfh.ch/projects/0001> ;
        knora-base:isDeleted false .
      BIND(<http://www.knora.org/ontology/0001/anything#Thing> AS ?resClass)
      {
        ?resource ?prop ?value .
        ?prop rdfs:subPropertyOf* knora-base:hasValue .
      }
      ?value knora-base:attachedToUser ?creator ;
        knora-base:hasPermissions ?permissions ;
        knora-base:isDeleted false .
      FILTER (?prop != knora-base:hasStandoffLinkToValue)
    }
    FILTER NOT EXISTS {
      ?value a ?fileClass .
      ?fileClass rdfs:subClassOf* knora-base:FileValue .
    }
    FILTER (STR(?permissions) IN ("M knora-admin:ProjectMember", "V knora-admin:KnownUser|RV \"q\" knora-admin:UnknownUser"))
  }
  OPTIONAL { ?resource rdfs:label ?label . }
  BIND(COALESCE(?label, STR(?resource)) AS ?labelOrIri)
}
ORDER BY ASC(?labelOrIri) ASC(?resource)
LIMIT 25
OFFSET 50
