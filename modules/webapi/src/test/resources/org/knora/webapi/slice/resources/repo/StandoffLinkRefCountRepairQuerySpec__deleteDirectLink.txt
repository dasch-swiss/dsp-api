PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX knora-base: <http://www.knora.org/ontology/knora-base#>

DELETE {
  GRAPH <http://www.knora.org/data/0001/anything> {
    <http://rdfh.ch/0001/source> <http://www.knora.org/ontology/knora-base#hasStandoffLinkTo> <http://rdfh.ch/0001/target> .
  }
}
WHERE {
  GRAPH <http://www.knora.org/data/0001/anything> {
    <http://rdfh.ch/0001/source> a ?sourceClass ;
      knora-base:isDeleted false .
  }
  ?sourceClass rdfs:subClassOf* knora-base:Resource .
  FILTER NOT EXISTS {
    GRAPH <http://www.knora.org/data/0001/anything> {
      <http://rdfh.ch/0001/source> <http://www.knora.org/ontology/knora-base#hasStandoffLinkToValue> ?currentLinkValue .
      ?currentLinkValue knora-base:isDeleted false ;
        rdf:object <http://rdfh.ch/0001/target> .
    }
  }
}
