PREFIX knora-base: <http://www.knora.org/ontology/knora-base#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?resource ?labelOrIri
WHERE {
  {
    SELECT DISTINCT ?resource
    WHERE {
      {
        {
          SELECT ?resource
          WHERE {
            VALUES ?permissions { "M knora-admin:ProjectMember" "V knora-admin:KnownUser|RV \"q\" knora-admin:UnknownUser" }
            GRAPH <http://www.knora.org/data/0001/anything> { ?resource knora-base:hasPermissions ?permissions . }
          }
        }
        ?resource a <http://www.knora.org/ontology/0001/anything#Thing> ;
          knora-base:isDeleted false .
      }
    }
  }
  OPTIONAL { ?resource rdfs:label ?label . }
  BIND(COALESCE(?label, STR(?resource)) AS ?labelOrIri)
}
ORDER BY ASC(?labelOrIri) ASC(?resource)
LIMIT 25
OFFSET 50
