SELECT DISTINCT ?thing (GROUP_CONCAT(DISTINCT(IF(BOUND(?text), STR(?text), "")); SEPARATOR='') AS ?text__Concat)
WHERE {
?text <http://jena.apache.org/text#query> ("interesting text" 1000000) .
?thing <http://www.knora.org/ontology/0001/anything#hasRichtext> ?text .
?text <http://www.knora.org/ontology/knora-base#valueHasStandoff> ?standoffTag .
?standoffTag <http://www.knora.org/ontology/knora-base#standoffTagHasEnd> ?standoffTag__end .
?standoffTag <http://www.knora.org/ontology/knora-base#standoffTagHasStart> ?standoffTag__start .
?text <http://www.knora.org/ontology/knora-base#valueHasString> ?text__valueHasString .
?standoffTag <http://www.w3.org/1999/02/22-rdf-syntax-ns#type> <http://www.knora.org/ontology/standoff#StandoffItalicTag> .
FILTER(regex(substr(?text__valueHasString, ?standoffTag__start + 1, ?standoffTag__end - ?standoffTag__start), "interesting", "i"))
FILTER(regex(substr(?text__valueHasString, ?standoffTag__start + 1, ?standoffTag__end - ?standoffTag__start), "text", "i"))
FILTER NOT EXISTS {
 ?thing <http://www.knora.org/ontology/knora-base#isDeleted> "true"^^<http://www.w3.org/2001/XMLSchema#boolean> .

}
FILTER NOT EXISTS {
 ?text <http://www.knora.org/ontology/knora-base#isDeleted> "true"^^<http://www.w3.org/2001/XMLSchema#boolean> .

}
}
GROUP BY ?thing
ORDER BY ASC(?thing)
LIMIT 25
