SELECT DISTINCT ?thing
WHERE {
{
?thing <http://www.knora.org/ontology/0001/anything#hasText> ?text .
?thing <http://www.knora.org/ontology/0001/anything#hasInteger> ?int .
?int <http://www.knora.org/ontology/knora-base#valueHasInteger> ?intVal .
FILTER((?intVal > "1"^^<http://www.w3.org/2001/XMLSchema#integer>))
FILTER NOT EXISTS {
 ?thing <http://www.knora.org/ontology/knora-base#isDeleted> "true"^^<http://www.w3.org/2001/XMLSchema#boolean> .

}
FILTER NOT EXISTS {
 ?text <http://www.knora.org/ontology/knora-base#isDeleted> "true"^^<http://www.w3.org/2001/XMLSchema#boolean> .

}
FILTER NOT EXISTS {
 ?int <http://www.knora.org/ontology/knora-base#isDeleted> "true"^^<http://www.w3.org/2001/XMLSchema#boolean> .

}
} UNION {
?thing <http://www.knora.org/ontology/0001/anything#hasRichtext> ?richtext .
FILTER NOT EXISTS {
 ?richtext <http://www.knora.org/ontology/knora-base#isDeleted> "true"^^<http://www.w3.org/2001/XMLSchema#boolean> .

}
}
}
GROUP BY ?thing
ORDER BY ASC(?thing)
LIMIT 25
