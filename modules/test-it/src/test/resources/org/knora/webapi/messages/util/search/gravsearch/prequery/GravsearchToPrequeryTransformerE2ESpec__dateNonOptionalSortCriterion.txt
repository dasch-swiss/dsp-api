SELECT DISTINCT ?thing (GROUP_CONCAT(DISTINCT(IF(BOUND(?date), STR(?date), "")); SEPARATOR='') AS ?date__Concat)
WHERE {
?date <http://www.knora.org/ontology/knora-base#valueHasStartJDN> ?date__valueHasStartJDN .
?thing <http://www.knora.org/ontology/0001/anything#hasDate> ?date .
FILTER NOT EXISTS {
 ?thing <http://www.knora.org/ontology/knora-base#isDeleted> "true"^^<http://www.w3.org/2001/XMLSchema#boolean> .

}
FILTER NOT EXISTS {
 ?date <http://www.knora.org/ontology/knora-base#isDeleted> "true"^^<http://www.w3.org/2001/XMLSchema#boolean> .

}
}
GROUP BY ?thing ?date__valueHasStartJDN
ORDER BY DESC(?date__valueHasStartJDN) ASC(?thing)
LIMIT 25
